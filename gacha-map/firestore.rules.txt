rules_version = '2';
// ガチャマップ Firestore セキュリティルール
// ・だれでも見られる（読み取りのみ）
// ・書き込めるのは「メール確認が済んだログイン中のユーザー」だけ
// ・項目の種類・文字数・日本国内の位置を、サーバー側でもチェック
// ・同じお店への書き込みは3秒以上あける（連打・荒らし対策）
// ・お店の削除は登録した本人だけ
service cloud.firestore {
  match /databases/{database}/documents {

    function verified() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.firebase.sign_in_provider == 'password';
    }

    function str(v, maxLen) {
      return v is string && v.size() <= maxLen;
    }

    // URL・メールアドレス・電話番号は名前に入れられない
    function safeName(v) {
      return !v.matches('.*(https?:|www[.]|@|0[0-9]{1,4}-[0-9]{1,4}-[0-9]{3,4}|0[789]0[0-9]{8}).*');
    }

    function validStore(d) {
      return d.keys().hasOnly(['name','type','lat','lng','hours','addr','osm','machines',
                               'by','createdAt','updatedAt','updatedBy'])
        && d.keys().hasAll(['name','type','lat','lng','hours','addr','machines',
                            'by','createdAt','updatedAt','updatedBy'])
        && str(d.name, 60) && d.name.size() >= 2 && safeName(d.name)
        && d.type in ['shop','game','mall','super','drug','books','station']
        && d.lat is number && d.lat >= 20 && d.lat <= 46
        && d.lng is number && d.lng >= 122 && d.lng <= 154
        && str(d.hours, 40) && str(d.addr, 80)
        && (!('osm' in d) || (str(d.osm, 30) && d.osm.matches('^(node|way|relation)/[0-9]+$')))
        && d.machines is list && d.machines.size() <= 300
        && d.updatedAt == request.time
        && d.updatedBy == request.auth.uid;
    }

    match /stores/{storeId} {
      allow read: if true;

      allow create: if verified()
        && storeId.matches('^[A-Za-z0-9_-]{1,40}$')
        && validStore(request.resource.data)
        && request.resource.data.by == request.auth.uid
        && request.resource.data.createdAt == request.time;

      allow update: if verified()
        && validStore(request.resource.data)
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['machines','hours','addr','updatedAt','updatedBy'])
        && request.time > resource.data.updatedAt + duration.value(3, 's');

      allow delete: if verified() && resource.data.by == request.auth.uid;
    }

    function today() {
      return math.floor(request.time.toMillis() / 86400000);
    }
    function storeExists(id) {
      return exists(/databases/$(database)/documents/stores/$(id));
    }

    // ニックネーム（だれでも見られる・本人だけ書ける・メールアドレスは保存しない）
    match /users/{uid} {
      allow read: if true;
      allow create, update: if verified() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['name','hideRank','updatedAt'])
        && (!('hideRank' in request.resource.data) || request.resource.data.hideRank is bool)
        && str(request.resource.data.name, 20) && request.resource.data.name.size() >= 1
        && !request.resource.data.name.matches('.*[<>@].*')
        && safeName(request.resource.data.name)
        && request.resource.data.updatedAt == request.time;
      allow delete: if request.auth != null && request.auth.uid == uid;
    }

    // 在庫報告の記録（ポイントの元）：同じお店で1人1日1回まで。本人だけ見られる。消せない。
    match /events/{eid} {
      allow read: if request.auth != null && resource.data.uid == request.auth.uid;
      allow create: if verified()
        && request.resource.data.keys().hasOnly(['uid','type','storeId','day','at'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.type == 'report'
        && request.resource.data.day == today()
        && request.resource.data.at == request.time
        && storeExists(request.resource.data.storeId)
        && eid == request.auth.uid + '_' + request.resource.data.storeId + '_' + string(request.resource.data.day);
      allow update, delete: if false;
    }

    // 写真：圧縮したJPEGだけ・1店1人1日3枚まで・削除は本人だけ・書き換え不可
    match /photos/{pid} {
      allow read: if true;
      allow create: if verified()
        && request.resource.data.keys().hasOnly(['by','storeId','img','w','h','day','n','createdAt'])
        && request.resource.data.by == request.auth.uid
        && request.resource.data.day == today()
        && request.resource.data.n in [1, 2, 3]
        && request.resource.data.w is int && request.resource.data.w > 0 && request.resource.data.w <= 1200
        && request.resource.data.h is int && request.resource.data.h > 0 && request.resource.data.h <= 1200
        && request.resource.data.img is string && request.resource.data.img.size() <= 300000
        && request.resource.data.img.matches('^data:image/jpeg;base64,[A-Za-z0-9+/]+=*$')
        && request.resource.data.createdAt == request.time
        && storeExists(request.resource.data.storeId)
        && pid == request.auth.uid + '_' + request.resource.data.storeId + '_'
                  + string(request.resource.data.day) + '_' + string(request.resource.data.n);
      allow delete: if request.auth != null && resource.data.by == request.auth.uid;
      allow update: if false;
    }

    // いいね：1人1回・自分の投稿には不可・取り消しは本人だけ
    match /likes/{lid} {
      allow read: if true;
      allow create: if verified()
        && request.resource.data.keys().hasOnly(['uid','target','tid','owner','storeId','at'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.at == request.time
        && request.resource.data.target in ['store','photo']
        && lid == request.auth.uid + '_' + request.resource.data.target + '_' + request.resource.data.tid
        && request.resource.data.owner != request.auth.uid
        && (
          (request.resource.data.target == 'store'
            && request.resource.data.tid == request.resource.data.storeId
            && get(/databases/$(database)/documents/stores/$(request.resource.data.tid)).data.by == request.resource.data.owner)
          ||
          (request.resource.data.target == 'photo'
            && get(/databases/$(database)/documents/photos/$(request.resource.data.tid)).data.by == request.resource.data.owner
            && get(/databases/$(database)/documents/photos/$(request.resource.data.tid)).data.storeId == request.resource.data.storeId)
        );
      allow delete: if request.auth != null && resource.data.uid == request.auth.uid;
      allow update: if false;
    }

    // 写真の通報：1人1枚に1回。3件で画面から非表示
    match /reports/{rid} {
      allow read: if true;
      allow create: if verified()
        && request.resource.data.keys().hasOnly(['uid','pid','storeId','at'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.at == request.time
        && rid == request.auth.uid + '_' + request.resource.data.pid
        && get(/databases/$(database)/documents/photos/$(request.resource.data.pid)).data.storeId == request.resource.data.storeId;
      allow update, delete: if false;
    }

    // お店の通報：1人1店1回・自分が登録したお店は不可・取り消しは本人だけ。3件で画面から非表示
    match /storeReports/{rid} {
      allow read: if true;
      allow create: if verified()
        && request.resource.data.keys().hasOnly(['uid','sid','reason','at'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.at == request.time
        && request.resource.data.reason in ['closed','wrong','bad']
        && rid == request.auth.uid + '_' + request.resource.data.sid
        && get(/databases/$(database)/documents/stores/$(request.resource.data.sid)).data.by != request.auth.uid;
      allow delete: if request.auth != null && resource.data.uid == request.auth.uid;
      allow update: if false;
    }

    // それ以外の場所は、だれも読み書きできない
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
